cmake_minimum_required(VERSION 3.22)
project(capy_offline_voice LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT EXISTS "${CAPY_WHISPER_SOURCE}/include/whisper.h")
  message(FATAL_ERROR "Supply the separately verified pinned Whisper source")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(flag WHISPER_BUILD_TESTS WHISPER_BUILD_EXAMPLES WHISPER_BUILD_SERVER WHISPER_CURL WHISPER_USE_SYSTEM_GGML WHISPER_USE_SYSTEM_LLAMA GGML_NATIVE GGML_AVX GGML_AVX2 GGML_AVX512 GGML_FMA GGML_F16C GGML_OPENMP GGML_BACKEND_DL GGML_RPC GGML_CUDA GGML_VULKAN GGML_METAL)
  set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${CAPY_WHISPER_SOURCE}" whisper)
add_library(capy_offline_voice STATIC offline_engine.cpp)
target_link_libraries(capy_offline_voice PUBLIC whisper)
target_include_directories(capy_offline_voice PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(ANDROID)
  add_library(capy_voice_jni SHARED android_jni.cpp)
  target_link_libraries(capy_voice_jni PRIVATE capy_offline_voice)
else()
  add_executable(capy_voice_runtime_test offline_engine_test.cpp)
  target_link_libraries(capy_voice_runtime_test PRIVATE capy_offline_voice)
endif()
