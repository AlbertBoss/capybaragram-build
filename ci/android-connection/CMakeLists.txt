cmake_minimum_required(VERSION 3.22)
project(capy_android_connection LANGUAGES C CXX)
if(NOT ANDROID OR NOT ANDROID_ABI MATCHES "^(arm64-v8a|x86_64)$")
    message(FATAL_ERROR "This JNI target supports the verified Android 64-bit ABIs only")
endif()
if(NOT EXISTS "${CAPY_CONNECTION_LIBRARY}" OR NOT EXISTS "${CAPY_CONNECTION_HEADER_DIR}/native_api.h")
    message(FATAL_ERROR "Supply the separately prepared and verified native transport")
endif()
add_library(capy_android_transport STATIC IMPORTED)
set_target_properties(capy_android_transport PROPERTIES IMPORTED_LOCATION "${CAPY_CONNECTION_LIBRARY}")
add_library(capy_connection_jni SHARED android_jni.cpp)
target_compile_features(capy_connection_jni PRIVATE cxx_std_17)
target_include_directories(capy_connection_jni PRIVATE "${CAPY_CONNECTION_HEADER_DIR}")
target_compile_options(capy_connection_jni PRIVATE -Wall -Wextra -Werror -Wconversion -fvisibility=hidden)
target_link_options(capy_connection_jni PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL" "-Wl,--no-undefined")
target_link_libraries(capy_connection_jni PRIVATE capy_android_transport log dl m)
