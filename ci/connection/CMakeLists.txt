cmake_minimum_required(VERSION 3.22)
project(capy_connection_native_probe LANGUAGES CXX)
if(NOT EXISTS "${CAPY_CONNECTION_LIBRARY}")
  message(FATAL_ERROR "Pinned Rust library missing")
endif()
add_executable(capy_connection_native_probe native_api_probe.cpp)
target_compile_features(capy_connection_native_probe PRIVATE cxx_std_17)
target_link_libraries(capy_connection_native_probe PRIVATE "${CAPY_CONNECTION_LIBRARY}")
if(WIN32 AND MSVC)
  target_compile_options(capy_connection_native_probe PRIVATE /W4 /WX)
  target_link_libraries(capy_connection_native_probe PRIVATE ws2_32 crypt32 secur32 ncrypt bcrypt userenv advapi32 ntdll)
elseif(UNIX)
  find_package(OpenSSL REQUIRED)
  find_package(Threads REQUIRED)
  target_compile_options(capy_connection_native_probe PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(capy_connection_native_probe PRIVATE OpenSSL::SSL OpenSSL::Crypto Threads::Threads ${CMAKE_DL_LIBS} m)
else()
  message(FATAL_ERROR "Unverified native ABI toolchain")
endif()
